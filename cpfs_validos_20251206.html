SELECT
    x.username,
    x.client_version,
    COUNT(*) AS qtd
FROM (
    SELECT
        s.username,
        s.program,
        s.machine,
        s.osuser,
        MAX(sci.client_driver)  AS client_driver,
        MAX(sci.client_version) AS client_version
    FROM
        v$session s
    LEFT JOIN v$session_connect_info sci
           ON sci.sid = s.sid
          AND sci.serial# = s.serial#
    WHERE
        s.username IS NOT NULL
    -- AND s.username = 'ADMNFA001'
    -- AND s.program LIKE '%exe%'
    -- AND s.machine LIKE '%sfz%'
    GROUP BY
        s.username,
        s.program,
        s.machine,
        s.osuser
) x
WHERE
    x.client_version < '12'
-- AND x.client_driver LIKE '%PHP%'
GROUP BY
    x.username,
    x.client_version
ORDER BY
	qtd DESC,
    x.username,
    x.client_version
